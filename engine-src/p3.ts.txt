      hazards.push({
        kind: "chip",
        x,
        y: y + (i - 1) * 16,
        w: 13,
        h: 13,
        alive: true,
        checked: false,
      });
    }
    safeX = origin + offsets[skip]!;
    safeX = clamp(safeX, 30, VW - 30);
  }

  function spawnGems(y: number) {
    const cx = clamp(safeX + (Math.random() * 36 - 18), 40, VW - 40);
    for (let i = -1; i <= 1; i++) {
      gemList.push({
        x: cx + i * 28,
        y: y - i * 10,
        alive: true,
        spin: Math.random(),
      });
    }
  }

  function spawn(diff: number) {
    const y = -28;
    if (distance < 560) {
      spawnGems(y);
      return;
    }
    const r = Math.random();
    if (r < 0.46) spawnGate(y, diff);
    else if (r < 0.74) spawnBlock(y, diff);
    else if (r < 0.9) spawnChips(y, diff);
    else spawnGems(y);
  }

  function circleHitsRect(cx: number, cy: number, r: number, x: number, y: number, w: number, h: number) {
    const nx = clamp(cx, x, x + w);
    const ny = clamp(cy, y, y + h);
    const dx = cx - nx;
    const dy = cy - ny;
    return dx * dx + dy * dy < r * r;
  }

  function collideHazard(h: Hazard) {
    if (!h.alive) return;
    let hit = false;
    if (h.kind === "chip") {
      const dx = playerX - h.x;
      const dy = PLAYER_Y - h.y;
      const rr = HIT_R + h.w;
      hit = dx * dx + dy * dy < rr * rr;
    } else {
      hit = circleHitsRect(playerX, PLAYER_Y, HIT_R, h.x, h.y, h.w, h.h);
    }
    if (!hit || invuln > 0) return;
    if (h.kind === "chip" && face().smash) {
      h.alive = false;
      bumpCombo();
      addScore(16, "smash");
      pop = 1;
      burst(h.x, h.y, CHIP, 8, 140);
      audio.blip(180, 90, 0.07, 0.06);
      trauma = Math.min(1, trauma + 0.18);
      return;
    }
    if (ward) {
      ward = false;
      h.alive = false;
      invuln = 0.85;
      hitstop = 0.07;
      trauma = Math.min(1, trauma + 0.55);
      flash = 0.45;
      pop = 1;
      burst(playerX, PLAYER_Y, TEAL, 16, 180);
      audio.ward();
      floaters.push({
        x: playerX,
        y: PLAYER_Y - 36,
        text: "ward",
        life: 0.7,
        max: 0.7,
        color: TEAL,
      });
      return;
    }
    kill();
  }

  function nearMiss(h: Hazard) {
    if (!h.alive || h.checked) return;
    const cy = h.kind === "chip" ? h.y : h.y + h.h / 2;
    if (cy < PLAYER_Y) return;
    h.checked = true;
    let clear: number;
    if (h.kind === "chip") clear = Math.abs(playerX - h.x) - (HIT_R + h.w);
    else if (playerX < h.x) clear = h.x - playerX - HIT_R;
    else if (playerX > h.x + h.w) clear = playerX - (h.x + h.w) - HIT_R;
    else clear = -1;
    if (clear >= 0 && clear < 34) {
      nears += 1;
      bumpCombo();
      addScore(22, "close");
      burst(playerX, PLAYER_Y, PAPER, 6, 80);
      audio.blip(720, 980, 0.06, 0.035);
    }
  }

  function collectGem(g: Gem, x: number, y: number) {
    if (!g.alive) return;
    const dx = g.x - x;
    const dy = g.y - y;
    if (dx * dx + dy * dy > (HIT_R + 11) * (HIT_R + 11)) return;
    g.alive = false;
    gems += 1;
    bumpCombo();
    addScore(10, "points");
    pop = 1;
    burst(g.x, g.y, TEAL, 8, 110);
    audio.blip(660, 1180, 0.07, 0.05);
  }

  function updatePlay(dt: number) {
    if (dying > 0) {
      dying -= dt;
      if (dying <= 0) finishDeath();
      return;
    }
    runT += dt;
    seconds += dt;
    hint = Math.max(0, hint - dt);
    invuln = Math.max(0, invuln - dt);
    comboT -= dt;
    if (comboT <= 0) combo = 0;

    updateFace(dt);
    updatePlayer(dt);
    updateEcho(dt);

    const diff = difficulty();
    const scroll = scrollSpeed();
    distance += scroll * dt;
    score += scroll * dt * 0.05 * face().score;
    scrollMeter += scroll * dt;
    if (scrollMeter >= spawnSpacing(diff)) {
      scrollMeter = 0;
      spawn(diff);
    }

    for (const h of hazards) {
      h.y += scroll * dt;
      collideHazard(h);
      nearMiss(h);
    }
    for (let i = hazards.length - 1; i >= 0; i--) {
      const h = hazards[i]!;
      if (!h.alive || h.y > VH + 80) hazards.splice(i, 1);
    }

    const ghostX = echo[0] ?? playerX;
    for (const g of gemList) {
      if (!g.alive) continue;
      g.y += scroll * dt;
      g.spin += dt * 2.4;
      if (face().magnet) {
        const dx = playerX - g.x;
        const dy = PLAYER_Y - g.y;
        const dist = Math.hypot(dx, dy);
        if (dist < 168 && dist > 0.001) {
          g.x += (dx / dist) * 240 * dt;
          g.y += (dy / dist) * 240 * dt;
        }
      }
      collectGem(g, playerX, PLAYER_Y);
      if (face().echo) collectGem(g, ghostX, PLAYER_Y);
    }
    for (let i = gemList.length - 1; i >= 0; i--) {
      const g = gemList[i]!;
      if (!g.alive || g.y > VH + 40) gemList.splice(i, 1);
    }
  }
